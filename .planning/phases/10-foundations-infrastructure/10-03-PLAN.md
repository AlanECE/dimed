```yaml
services:
  db:
    image: postgres:16-slim
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-dimed}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-dimed}
      POSTGRES_DB: ${POSTGRES_DB:-dimed}
    ports:
      - "5432:5432"
    volumes:
      - pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U dimed"]
      interval: 5s
      timeout: 5s
      retries: 5

  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 5s
      retries: 5

  api:
    build:
      context: ..
      dockerfile: docker/Dockerfile.api
      target: dev
    ports:
      - "8000:8000"
    volumes:
      - ../apps/api:/app
      - ../packages/ocr:/packages/ocr
    environment:
      DIMED_DATABASE_URL: ${DIMED_DATABASE_URL:-postgresql+asyncpg://dimed:dimed@db:5432/dimed}
      DIMED_REDIS_URL: ${DIMED_REDIS_URL:-redis://redis:6379/0}
      DIMED_SECRET_KEY: ${DIMED_SECRET_KEY:-dev-secret-change-in-production}
    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_healthy
    command: uvicorn app.main:app --host 0.0.0.0 --port 8000 --reload

  web:
    build:
      context: ..
      dockerfile: docker/Dockerfile.web
      target: dev
    ports:
      - "3000:3000"
    volumes:
      - ../apps/web:/app
      - ../packages/shared-types:/packages/shared-types
      - ../packages/ui:/packages/ui
      - /app/node_modules
    depends_on:
      - api
    command: pnpm dev

volumes:
  pgdata:
```

<task id="03.2">
<title>Create API Dockerfile (dev + prod multi-stage)</title>
<read_first>
- apps/api/pyproject.toml (dependencies)
- docker/docker-compose.yml (context and target)
</read_first>
<action>
`docker/Dockerfile.api`:
```dockerfile
# ---- Base ----
FROM python:3.12-slim AS base
WORKDIR /app
RUN pip install --no-cache-dir uv

# ---- Dev ----
FROM base AS dev
COPY apps/api/pyproject.toml .
RUN uv pip install --system -e .
# Source mounted via volume in docker-compose

# ---- Builder (prod) ----
FROM base AS builder
COPY apps/api/pyproject.toml .
COPY packages/ocr /packages/ocr
RUN uv pip install --system . && \
    uv pip install --system -e /packages/ocr

COPY apps/api/ .

# ---- Production ----
FROM python:3.12-slim AS production
WORKDIR /app

RUN groupadd -r dimed && useradd -r -g dimed -d /app -s /sbin/nologin dimed

COPY --from=builder /usr/local/lib/python3.12/site-packages /usr/local/lib/python3.12/site-packages
COPY --from=builder /usr/local/bin/uvicorn /usr/local/bin/uvicorn
COPY --from=builder /usr/local/bin/alembic /usr/local/bin/alembic
COPY --from=builder /app /app

USER dimed
EXPOSE 8000

HEALTHCHECK --interval=30s --timeout=10s --start-period=5s --retries=3 \
    CMD python -c "import urllib.request; urllib.request.urlopen('http://localhost:8000/health')"

CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000"]
```
</action>
<acceptance_criteria>
- `docker/Dockerfile.api` contains `FROM python:3.12-slim AS base`
- `docker/Dockerfile.api` contains `FROM base AS dev`
- `docker/Dockerfile.api` contains `FROM python:3.12-slim AS production`
- `docker/Dockerfile.api` contains `USER dimed` (non-root)
- `docker/Dockerfile.api` contains `HEALTHCHECK`
</acceptance_criteria>

<task id="03.3">
<title>Create Web Dockerfile (dev + prod multi-stage)</title>
<read_first>
- apps/web/package.json (dependencies)
- docker/docker-compose.yml (context and target)
</read_first>
<action>
`docker/Dockerfile.web`:
```dockerfile
# ---- Base ----
FROM node:22-slim AS base
RUN corepack enable && corepack prepare pnpm@9.15.0 --activate
WORKDIR /app

# ---- Dev ----
FROM base AS dev
COPY apps/web/package.json .
RUN pnpm install
# Source mounted via volume in docker-compose

# ---- Builder (prod) ----
FROM base AS builder
COPY package.json pnpm-workspace.yaml pnpm-lock.yaml ./
COPY apps/web/ ./apps/web/
COPY packages/shared-types/ ./packages/shared-types/
COPY packages/ui/ ./packages/ui/
RUN pnpm install --frozen-lockfile
RUN pnpm --filter @dimed/web build

# ---- Production ----
FROM node:22-slim AS production
WORKDIR /app

RUN groupadd -r dimed && useradd -r -g dimed -d /app -s /sbin/nologin dimed

COPY --from=builder /app/apps/web/.next/standalone ./
COPY --from=builder /app/apps/web/.next/static ./.next/static
COPY --from=builder /app/apps/web/public ./public

USER dimed
EXPOSE 3000

HEALTHCHECK --interval=30s --timeout=10s --start-period=5s --retries=3 \
    CMD node -e "require('http').get('http://localhost:3000', (r) => process.exit(r.statusCode === 200 ? 0 : 1))"

CMD ["node", "server.js"]
```
</action>

<task id="03.4">
<title>Add startup event for DB connection check</title>
<read_first>
- apps/api/app/main.py (current state)
- apps/api/app/db/session.py (engine)
</read_first>
<action>
Update `apps/api/app/main.py` to add a lifespan that tests DB connectivity on startup:

```python
from contextlib import asynccontextmanager
from fastapi import FastAPI
from app.db.session import engine

@asynccontextmanager
async def lifespan(app: FastAPI):
    # Verify DB connection on startup
    async with engine.begin() as conn:
        await conn.execute(text("SELECT 1"))
    yield
    await engine.dispose()

app = FastAPI(
    title="DIMED API",
    version="0.1.0",
    description="Pharmaceutical logistics management",
    lifespan=lifespan,
)

@app.get("/health")
async def health():
    return {"status": "ok", "service": "dimed-api"}
```

Add `from sqlalchemy import text` import.
</action>
<acceptance_criteria>
- `apps/api/app/main.py` contains `async def lifespan`
- `apps/api/app/main.py` contains `lifespan=lifespan`
- `apps/api/app/main.py` contains `SELECT 1`
- `apps/api/app/main.py` contains `engine.dispose()`
</acceptance_criteria>
</task>
